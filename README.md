# Проект для автоматического сбора информации о книгах с сайта Books to Scrape.
# Парсер извлекает данные со всех страниц каталога и сохраняет их в структурированном формате.

# Структура проекта:
text
books_scraper/
├── artifacts/
│   └── books_data.txt              # Сохраненные данные книг
├── notebooks/
│   └── Писарская Д.Д._HW_03_python_ds_2025.ipynb  # Jupyter ноутбук с анализом
├── scraper.py                      # Основной скрипт парсера
├── README.md                       # Документация
├── tests/
│   └── test_scraper.py             # Автотесты
├── .gitignore                      # Игнорируемые файлы
└── requirements.txt                # Зависимости

# Возможности:
Парсинг данных книги - извлечение полной информации со страницы книги
Обход всего каталога - автоматический проход по всем страницам категорий
Автоматизация - ежедневный запуск в 19:00 с помощью библиотеки schedule
Тестирование - автотесты для ключевых функций парсинга
Сохранение данных - запись результатов в текстовый файл

# Создание виртуального окружения
